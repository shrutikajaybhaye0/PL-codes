Q1 Write a program to accept a student's name, roll number, and branch, then display the details.

#include<stdio.h>
#include<string.h>
int main () {
char a[50];
int b;
char c[50];
printf("enter student name:");
scanf("%s",&a);
printf("enter roll number:");
scanf("%d",&b);
printf("enter branch:");
scanf("%s",&c);
printf("name of student is = %s\n",a);
printf("student roll number is = %d\n",b);
printf("branch of student is = %s\n",c);
return 0;
}

Q2 Write a program to accept two integers and display their sum, difference, product, and quotient.

printf("a*b = %d\n",a*b);
